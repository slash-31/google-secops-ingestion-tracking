Legacy BigQuery Query (Historical Reference):
SELECT log_type,
    string_agg(DISTINCT collector_id, ':RQ:') as collector_ids, 
    string_agg(DISTINCT input_type, ':RQ:') as input_types, 
    string_agg(DISTINCT drop_reason_code, ':RQ:') as drop_reason_codes,
    SUM(CASE WHEN component = 'Ingestion API' THEN log_volume /1000/1000 END) as size_mb,
    SUM(CASE WHEN component = 'Normalizer' THEN event_count END) as event_count,
    SUM(CASE WHEN component = 'Normalizer' AND state = 'validated' THEN event_count END) as normalized_events,
    SUM(CASE WHEN component = 'Normalizer' AND state = 'failed_validation' THEN event_count WHEN component = 'Normalizer' AND state = 'failed_parsing' THEN log_count END) as error_events,
    SUM(CASE WHEN component = 'Normalizer' AND state = 'failed_parsing' THEN log_count END) as parsing_error_events,
    SUM(CASE WHEN component = 'Normalizer' AND state = 'failed_validation' THEN event_count END) as validation_error_events
FROM `chronicle-{{ project_id }}.datalake.ingestion_metrics`
WHERE UNIX_SECONDS(TIMESTAMP (ingestion_metrics.start_time)) > @start_epoch 
    AND UNIX_SECONDS(TIMESTAMP (ingestion_metrics.start_time)) < @end_epoch
GROUP BY log_type
ORDER BY size_mb DESC, event_count DESC;
